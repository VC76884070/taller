-- por cada orden de trabajo
UPDATE avance_trabajo a
SET numero_avance = subquery.numero
FROM (
    SELECT 
        id,
        ROW_NUMBER() OVER (
            PARTITION BY id_orden_trabajo 
            ORDER BY fecha_creacion
        ) as numero
    FROM avance_trabajo
) subquery
WHERE a.id = subquery.id;
